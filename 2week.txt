        function parseData(dataArray) {
            return dataArray.map(str => {
                if (str.startsWith("OX ")) {
                    const parts = str.split('|');
                    const idText = parts[0].replace("OX ", "").trim();
                    const idMatch = idText.match(/^(\d+)\s(.*)/);
                    return {
                        id: parseInt(idMatch[1]),
                        type: 'ox',
                        text: idMatch[2],
                        answer: parts[1],
                        explanation: parts[2] || '정답 확인'
                    };
                } else {
                    const idMatch = str.match(/^(\d+)\s(.*)/);
                    if(!idMatch) return null;
                    const id = parseInt(idMatch[1]);
                    let text = idMatch[2];
                    let blanks = {};
                    let groupOrder = [];
                    
                    text = text.replace(/\[(.*?)\]/g, (match, p1) => {
                        const answers = p1.split('|');
                        const primaryAns = answers[0];
                        // 그룹 키를 통해 같은 정답을 가진 빈칸들을 묶음
                        const groupKey = "g_" + primaryAns.replace(/\s+/g, '');
                        
                        if (!blanks[groupKey]) {
                            blanks[groupKey] = answers;
                            groupOrder.push(groupKey); // 처음 등장한 빈칸 그룹 순서 저장
                        }
                        
                        // 기호 할당 (①, ②, ③...)
                        const symbolIndex = groupOrder.indexOf(groupKey);
                        const symbol = circleNumbers[symbolIndex] || '';
                        
                        // 기호와 입력창이 줄바꿈으로 분리되지 않도록 inline-flex 적용
                        return `<span class="inline-flex items-center whitespace-nowrap"><span class="text-blue-600 font-bold ml-1 mr-0.5 select-none">${symbol}</span><input type="text" data-qid="${id}" data-group="${groupKey}" class="blank-input" autocomplete="off" oninput="syncBlanks(this, ${id})"></span>`;
                    });
                    
                    // 해설란에 기호와 함께 정답 표시 (예: ① 결핍 욕구, ② 안전 욕구)
                    const explanationText = groupOrder.map((key, idx) => `${circleNumbers[idx]} ${blanks[key][0]}`).join(', ');
                    
                    return { 
                        id, 
                        type: 'fill', 
                        html: text, 
                        blanks, 
                        explanation: '정답: ' + explanationText 
                    };
                }
            }).filter(item => item !== null);
        }

                sectionWrapper.className = `section-wrapper space-y-6 hidden`; // 기본적으로 모두 숨김
                
                // 섹션 헤더
                const sectionHeader = document.createElement('div');
                sectionHeader.className = "mb-8";
                sectionHeader.innerHTML = `
                    <div class="bg-gradient-to-r from-slate-800 to-slate-700 text-white px-5 py-4 md:px-8 md:py-6 rounded-3xl shadow-lg flex items-center justify-between border-l-[6px] border-blue-400">
                        <h2 class="text-xl md:text-2xl font-bold tracking-tight">${sectionTitle}</h2>
                        <span class="bg-white/20 backdrop-blur-sm text-white text-sm md:text-base px-4 py-1.5 rounded-full font-bold shadow-sm whitespace-nowrap ml-4">Q.${startId} ~ ${endId}</span>
                    </div>
                `;
                sectionWrapper.appendChild(sectionHeader);

                // 해당 섹션의 문제들만 필터링하여 렌더링
                const sectionQuestions = questions.filter(q => q.id >= startId && q.id <= endId);
                
                sectionQuestions.forEach((q) => {
                    const card = document.createElement('div');
                    card.className = "bg-white p-6 md:p-8 rounded-3xl shadow-sm border border-slate-200 transition-all hover:shadow-md";
                    card.id = `card-${q.id}`;
                    
                    let contentHTML = `<div class="flex items-start gap-4 md:gap-5 mb-5">
                        <span class="flex-shrink-0 bg-blue-500 text-white font-bold text-lg w-10 h-10 flex items-center justify-center rounded-xl shadow-sm">${q.id}</span>
                        <div class="text-lg leading-loose text-slate-800 flex-grow pt-1">`;

                    if (q.type === 'fill') {
                        contentHTML += q.html;
                    } else if (q.type === 'ox') {
                        contentHTML += `<p class="mb-5 font-medium">${q.text}</p>
                            <div class="flex gap-4 max-w-xs md:max-w-sm mx-auto md:mx-0">
                                <button onclick="selectOX(${q.id}, 'O')" id="btn-${q.id}-O" class="ox-btn flex-1 py-4 border-2 border-slate-200 rounded-2xl text-2xl font-bold text-slate-400 hover:border-blue-400 hover:text-blue-500 bg-slate-50">O</button>
                                <button onclick="selectOX(${q.id}, 'X')" id="btn-${q.id}-X" class="ox-btn flex-1 py-4 border-2 border-slate-200 rounded-2xl text-2xl font-bold text-slate-400 hover:border-blue-400 hover:text-blue-500 bg-slate-50">X</button>
                            </div>`;
                    }

                    contentHTML += `</div></div>
                        <div class="flex justify-end mb-2">
                            <button onclick="checkAnswer(${q.id})" id="submit-btn-${q.id}" class="bg-slate-800 hover:bg-slate-900 text-white font-semibold py-2.5 px-8 rounded-xl transition-colors shadow-sm">채점하기</button>
                        </div>
                        <div id="feedback-${q.id}" class="hidden p-5 rounded-2xl mt-4 transition-all duration-300">
                            <p id="feedback-msg-${q.id}" class="font-bold text-lg mb-2"></p>
                            <div class="h-px bg-current opacity-20 w-full mb-3"></div>
                            <p id="feedback-exp-${q.id}" class="text-slate-700 font-medium leading-relaxed"></p>
                        </div>`;

                    card.innerHTML = contentHTML;
                    sectionWrapper.appendChild(card);
                });

                // 섹션 하단에 다음 파트로 이동 버튼 추가
                if (nextId) {
                    const nextBtnWrapper = document.createElement('div');
                    nextBtnWrapper.className = "flex justify-center mt-10 pt-4";
                    nextBtnWrapper.innerHTML = `<button onclick="showSection(${nextId})" class="bg-blue-50 hover:bg-blue-100 border border-blue-200 text-blue-700 font-bold py-4 px-8 rounded-2xl shadow-sm transition-colors flex items-center gap-3 text-lg w-full md:w-auto justify-center">다음 파트로 이동하기 <span>➡️</span></button>`;
                    sectionWrapper.appendChild(nextBtnWrapper);
                }

                container.appendChild(sectionWrapper);
            });
            
            // 초기 렌더링 시 첫 번째 섹션 보여주기
            showSection(1);
        }

        function showSection(targetId) {
            // 모든 섹션 숨기기
            document.querySelectorAll('.section-wrapper').forEach(el => el.classList.add('hidden'));
            
            // 모든 탭 스타일 초기화
            document.querySelectorAll('.tab-btn').forEach(el => {
                el.classList.remove('bg-blue-600', 'text-white', 'shadow-md', 'transform', 'scale-[1.02]');
                el.classList.add('text-slate-600', 'bg-slate-50');
            });

            // 타겟 섹션 보이기
            const targetSection = document.getElementById(`section-${targetId}`);
            if(targetSection) targetSection.classList.remove('hidden');

            // 타겟 탭 하이라이트 및 스크롤 포커스
            const targetTab = document.getElementById(`tab-btn-${targetId}`);
            if(targetTab) {
                targetTab.classList.remove('text-slate-600', 'bg-slate-50');
                targetTab.classList.add('bg-blue-600', 'text-white', 'shadow-md', 'transform', 'scale-[1.02]');
                
                // 선택된 탭이 중앙에 오도록 스크롤 이동
                targetTab.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
            }
            
            // 페이지 최상단으로 부드럽게 스크롤
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        function syncBlanks(element, qId) {
            const group = element.dataset.group;
            const value = element.value;
            const allInputs = document.querySelectorAll(`input[data-qid="${qId}"][data-group="${group}"]`);
            allInputs.forEach(input => {
                if (input !== element) input.value = value;
            });
        }

        window.userAnswers = {};
        function selectOX(qId, choice) {
            window.userAnswers[qId] = choice;
            document.getElementById(`btn-${qId}-O`).classList.remove('selected');
            document.getElementById(`btn-${qId}-X`).classList.remove('selected');
            document.getElementById(`btn-${qId}-${choice}`).classList.add('selected');
        }

        function checkAnswer(qId) {
            const q = questions.find(x => x.id === qId);
            const feedbackDiv = document.getElementById(`feedback-${q.id}`);
            const feedbackMsg = document.getElementById(`feedback-msg-${q.id}`);
            const feedbackExp = document.getElementById(`feedback-exp-${q.id}`);
            const submitBtn = document.getElementById(`submit-btn-${q.id}`);
            
            let isCorrect = true;

            if (q.type === 'fill') {
                let hasEmpty = false;
                const blankKeys = Object.keys(q.blanks);
                
                blankKeys.forEach(key => {
                    const inputs = document.querySelectorAll(`input[data-qid="${q.id}"][data-group="${key}"]`);
                    const userAnswer = inputs[0].value.trim().replace(/\s+/g, ''); 
                    if (!userAnswer) hasEmpty = true;

                    // 허용된 정답 목록 (공백 제거)
                    const acceptedAnswers = q.blanks[key].map(a => a.replace(/\s+/g, ''));
                    
                    inputs.forEach(input => {
                        input.classList.remove('correct', 'incorrect');
                        input.disabled = true; 
                        
                        if (acceptedAnswers.includes(userAnswer)) {
                            input.classList.add('correct');
                        } else {
                            isCorrect = false;
                            input.classList.add('incorrect');
                            input.value = `${inputs[0].value} (정답: ${q.blanks[key][0]})`;
                        }
                    });
                });

                if (hasEmpty) {
                    alert("모든 빈칸을 채워주세요!");
                    document.querySelectorAll(`input[data-qid="${q.id}"]`).forEach(input => {
                        input.disabled = false;
                        input.classList.remove('correct', 'incorrect');
                        if (input.value.includes('(정답:')) input.value = '';
                    });
                    return;
                }

            } else if (q.type === 'ox') {
                const userAnswer = window.userAnswers[q.id];
                if (!userAnswer) {
                    alert("O 또는 X를 선택해주세요!");
                    return;
                }
                
                const btnO = document.getElementById(`btn-${q.id}-O`);
                const btnX = document.getElementById(`btn-${q.id}-X`);
                btnO.disabled = true;
                btnX.disabled = true;
                
                if (userAnswer === q.answer) {
                    document.getElementById(`btn-${q.id}-${userAnswer}`).classList.replace('selected', 'correct');
                } else {
                    isCorrect = false;
                    document.getElementById(`btn-${q.id}-${userAnswer}`).classList.replace('selected', 'incorrect');
                    document.getElementById(`btn-${q.id}-${q.answer}`).classList.add('correct');
                }
            }

            // 피드백 UI 업데이트
            feedbackDiv.classList.remove('hidden', 'bg-emerald-50', 'bg-rose-50');
            submitBtn.style.display = 'none';

            if (isCorrect) {
                feedbackDiv.classList.add('bg-emerald-50');
                feedbackMsg.className = "font-bold text-lg mb-2 text-emerald-700 flex items-center gap-2";
                feedbackMsg.innerHTML = "<span>🎉</span> 정답입니다!";
            } else {
                feedbackDiv.classList.add('bg-rose-50');
                feedbackMsg.className = "font-bold text-lg mb-2 text-rose-700 flex items-center gap-2";
                feedbackMsg.innerHTML = "<span>💡</span> 아쉽네요, 오답입니다.";
            }
            
            feedbackExp.innerHTML = `<strong>📖 해설:</strong><br>${q.explanation}`;
            
            // 진행률 업데이트
            completedCount++;
            updateProgress();
        }
        
        function updateProgress() {
            const progressText = document.getElementById('progress-text');
            const progressBar = document.getElementById('progress-bar');
            
            progressText.innerText = `${completedCount} / 181 완료`;
            const percentage = (completedCount / 181) * 100;
            progressBar.style.width = `${percentage}%`;
        }

        // Initialize Application
        window.onload = renderQuiz;
    </script>
</body>
</html>
