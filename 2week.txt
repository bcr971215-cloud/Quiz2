        // 아이패드 파싱 오류 완벽 해결 (substring 방식)
        function parseRawData() {
            const partsMap = new Map();
            rawData.forEach(str => {
                const firstPipe = str.indexOf('|');
                const secondPipe = str.indexOf('|', firstPipe + 1);
                
                if(firstPipe === -1 || secondPipe === -1) return; // 오류 방지 안전장치
                
                const idStr = str.substring(0, firstPipe);
                const partStr = str.substring(firstPipe + 1, secondPipe);
                const textContent = str.substring(secondPipe + 1);
                
                const id = parseInt(idStr);
                
                let isOX = false;
                let oxAnswer = "";
                let oxExp = "";
                let cleanText = textContent;

                if (textContent.startsWith('[OX:')) {
                    isOX = true;
                    const closeBracket = textContent.indexOf(']');
                    const oxData = textContent.substring(4, closeBracket);
                    const firstColon = oxData.indexOf(':');
                    
                    if (firstColon !== -1) {
                        oxAnswer = oxData.substring(0, firstColon);
                        oxExp = oxData.substring(firstColon + 1);
                    } else {
                        oxAnswer = oxData;
                    }
                    cleanText = textContent.substring(closeBracket + 1).trim();
                }

                let blanks = {};
                if (!isOX) {
                    const blankRegex = /{([^}]+)}/g;
                    let match;
                    let bIndex = 1;
                    while ((match = blankRegex.exec(cleanText)) !== null) {
                        const content = match[1];
                        const firstColon = content.indexOf(':');
                        
                        let gId = '';
                        let answerStr = '';
                        
                        if (firstColon !== -1) {
                            gId = content.substring(0, firstColon);
                            answerStr = content.substring(firstColon + 1);
                        } else {
                            gId = `temp_${bIndex++}`;
                            answerStr = content;
                        }
                        
                        blanks[gId] = answerStr.split('|').map(a => a.trim());
                    }
                }

                const question = {
                    id,
                    part: partStr,
                    type: isOX ? 'ox' : 'fill',
                    text: cleanText,
                    blanks: isOX ? null : blanks,
                    oxAnswer,
                    oxExp
                };
                questions.push(question);

                if (!partsMap.has(partStr)) partsMap.set(partStr, []);
                partsMap.get(partStr).push(question);
            });

            partsMap.forEach((qs, p) => parsedParts.push({ name: p, questions: qs }));
        }

        function renderTabs() {
            const container = document.getElementById('tabs-container');
            container.innerHTML = '';
            
            parsedParts.forEach((partObj, index) => {
                const btn = document.createElement('button');
                btn.className = `tab-btn px-4 py-2 rounded-full text-sm font-medium border border-gray-200 bg-white text-gray-600 hover:bg-blue-50 focus:outline-none ${index === currentTabIndex ? 'active' : ''}`;
                btn.innerText = partObj.name;
                btn.onclick = () => {
                    currentTabIndex = index;
                    renderQuiz(currentTabIndex);
                    updateTabsStyle();
                    btn.scrollIntoView({ behavior: 'smooth', block: 'nearest', inline: 'center' });
                };
                container.appendChild(btn);
            });
        }

        function updateTabsStyle() {
            const btns = document.querySelectorAll('.tab-btn');
            btns.forEach((btn, index) => {
                if (index === currentTabIndex) {
                    btn.classList.add('active');
                } else {
                    btn.classList.remove('active');
                }
            });
            
            document.getElementById('prev-btn').classList.toggle('hidden', currentTabIndex === 0);
            document.getElementById('next-btn').classList.toggle('hidden', currentTabIndex === parsedParts.length - 1);
            updateProgress();
        }

                    <span class="flex-shrink-0 bg-gray-100 text-blue-600 font-extrabold w-8 h-8 flex items-center justify-center rounded-lg shadow-inner">${q.id}</span>
                    <div class="text-[15px] md:text-lg leading-relaxed text-gray-800 flex-grow font-medium">`;

                if (q.type === 'fill') {
                    let textWithInputs = q.text;
                    const blankKeys = Object.keys(q.blanks);
                    
                    blankKeys.forEach(key => {
                        const possibleAnswersStr = q.blanks[key].join('|');
                        const searchStr = key.startsWith('temp_') ? `{${possibleAnswersStr}}` : `{${key}:${possibleAnswersStr}}`;
                        
                        const inputElementHTML = `
                            <span class="input-wrapper">
                                <span class="symbol-badge">${getSymbol(key)}</span>
                                <input type="text" data-qid="${q.id}" data-group="${key}" class="blank-input" autocomplete="off" autocorrect="off" autocapitalize="off" spellcheck="false" oninput="syncBlanks(this, ${q.id})">
                            </span>`;
                        
                        textWithInputs = textWithInputs.split(searchStr).join(inputElementHTML);
                    });
                    contentHTML += textWithInputs;
                } else if (q.type === 'ox') {
                    contentHTML += `<p class="mb-4">${q.text}</p>
                        <div class="flex gap-4">
                            <button onclick="selectOX(${q.id}, 'O')" id="btn-${q.id}-O" class="ox-btn flex-1 border-2 border-gray-200 rounded-lg text-2xl font-bold text-gray-400 hover:border-blue-400 hover:text-blue-500 bg-white">O</button>
                            <button onclick="selectOX(${q.id}, 'X')" id="btn-${q.id}-X" class="ox-btn flex-1 border-2 border-gray-200 rounded-lg text-2xl font-bold text-gray-400 hover:border-red-400 hover:text-red-500 bg-white">X</button>
                        </div>`;
                }

                contentHTML += `</div></div>
                    <div class="flex justify-end mb-2">
                        <button onclick="checkAnswer(${q.id})" id="submit-btn-${q.id}" class="bg-gray-800 hover:bg-gray-900 text-white font-semibold py-3 px-6 rounded-lg transition-colors w-full md:w-auto shadow-md">채점하기</button>
                    </div>
                    <div id="feedback-${q.id}" class="hidden p-4 rounded-lg mt-3 transition-all duration-300">
                        <p id="feedback-msg-${q.id}" class="font-bold mb-2 flex items-center gap-2"></p>
                        <p id="feedback-exp-${q.id}" class="text-gray-700 text-[14px] leading-relaxed"></p>
                    </div>`;

                card.innerHTML = contentHTML;
                container.appendChild(card);
            });
        }

        const symbolMap = ['①', '②', '③', '④', '⑤', '⑥', '⑦', '⑧', '⑨', '⑩'];
        function getSymbol(key) {
            if(key.startsWith('temp_')) return '★';
            const num = parseInt(key);
            if(!isNaN(num) && num > 0 && num <= 10) {
                return symbolMap[num - 1];
            }
            return key;
        }

        window.syncBlanks = function(element, qId) {
            const group = element.dataset.group;
            const value = element.value;
            const allInputs = document.querySelectorAll(`input[data-qid="${qId}"][data-group="${group}"]`);
            
            allInputs.forEach(input => {
                if (input !== element) {
                    input.value = value;
                }
            });
        }

        window.selectOX = function(qId, choice) {
            userAnswers[qId] = choice;
            document.getElementById(`btn-${qId}-O`).classList.remove('selected');
            document.getElementById(`btn-${qId}-X`).classList.remove('selected');
            document.getElementById(`btn-${qId}-${choice}`).classList.add('selected');
        }

        window.checkAnswer = function(qId) {
            const q = questions.find(x => x.id === qId);
            const feedbackDiv = document.getElementById(`feedback-${q.id}`);
            const feedbackMsg = document.getElementById(`feedback-msg-${q.id}`);
            const feedbackExp = document.getElementById(`feedback-exp-${q.id}`);
            const submitBtn = document.getElementById(`submit-btn-${q.id}`);
            
            let isCorrect = true;
            let explanationHTML = "<strong>[정답 해설]</strong><br>";

            if (q.type === 'fill') {
                let hasEmpty = false;
                const blankKeys = Object.keys(q.blanks);
                let answerDetails = [];
                
                blankKeys.forEach(key => {
                    const inputs = document.querySelectorAll(`input[data-qid="${q.id}"][data-group="${key}"]`);
                    const userAnswer = inputs[0].value.trim().replace(/\s+/g, ''); 
                    if (!userAnswer) hasEmpty = true;

                    const acceptedAnswers = q.blanks[key].map(a => a.replace(/\s+/g, ''));
                    const formattedAnsText = q.blanks[key].join(' 또는 ');
                    
                    answerDetails.push(`${getSymbol(key)} <span class='text-blue-600 font-bold'>${formattedAnsText}</span>`);

                    inputs.forEach(input => {
                        input.classList.remove('correct', 'incorrect');
                        if (acceptedAnswers.includes(userAnswer)) {
                            input.classList.add('correct');
                        } else {
                            isCorrect = false;
                            input.classList.add('incorrect');
                        }
                        input.disabled = true;
                    });
                });

                if (hasEmpty) {
                    alert("모든 빈칸을 채워주세요.");
                    document.querySelectorAll(`input[data-qid="${q.id}"]`).forEach(input => {
                        input.disabled = false;
                        input.classList.remove('correct', 'incorrect');
                    });
                    return;
                }
                explanationHTML += `<div class="mt-2 bg-white p-3 rounded border border-blue-100 shadow-sm">${answerDetails.join('<br>')}</div>`;

            } else if (q.type === 'ox') {
                const userAnswer = userAnswers[q.id];
                if (!userAnswer) {
                    alert("O 또는 X를 선택해주세요.");
                    return;
                }
                
                const btnO = document.getElementById(`btn-${q.id}-O`);
                const btnX = document.getElementById(`btn-${q.id}-X`);
                btnO.disabled = true;
                btnX.disabled = true;
                
                if (userAnswer === q.oxAnswer) {
                    document.getElementById(`btn-${q.id}-${userAnswer}`).classList.replace('selected', 'correct');
                } else {
                    isCorrect = false;
                    document.getElementById(`btn-${q.id}-${userAnswer}`).classList.replace('selected', 'incorrect');
                    document.getElementById(`btn-${q.id}-${q.oxAnswer}`).classList.add('correct');
                }
                
                explanationHTML += `<p class="mt-2 text-blue-700 font-bold">정답: ${q.oxAnswer}</p>`;
                if(q.oxExp) {
                    explanationHTML += `<p class="mt-1 text-gray-700">${q.oxExp}</p>`;
                }
            }

            feedbackDiv.classList.remove('hidden', 'bg-green-50', 'bg-red-50');
            submitBtn.style.display = 'none';

            if (isCorrect) {
                feedbackDiv.classList.add('bg-green-50', 'border', 'border-green-200');
                feedbackMsg.className = "font-bold flex items-center gap-2 text-green-700";
                feedbackMsg.innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" class="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 12l2 2 4-4m6 2a9 9 0 11-18 0 9 9 0 0118 0z" /></svg> 정답입니다!`;
            } else {
                feedbackDiv.classList.add('bg-red-50', 'border', 'border-red-200');
                feedbackMsg.className = "font-bold flex items-center gap-2 text-red-700";
                feedbackMsg.innerHTML = `<svg xmlns="http://www.w3.org/2000/svg" class="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M10 14l2-2m0 0l2-2m-2 2l-2-2m2 2l2 2m7-2a9 9 0 11-18 0 9 9 0 0118 0z" /></svg> 오답입니다. 해설을 확인하세요.`;
            }
            
            feedbackExp.innerHTML = explanationHTML;
            
            solvedQuestions.add(q.id);
            updateProgress();
        }

        function updateProgress() {
            const el = document.getElementById('progress-text');
            const total = questions.length;
            const solved = solvedQuestions.size;
            el.innerText = `완료: ${solved} / ${total} 문항`;
        }

        window.resetQuiz = function() {
            if(confirm("모든 풀이 내역을 완전히 초기화하시겠습니까?")) {
                solvedQuestions.clear();
                for (let key in userAnswers) delete userAnswers[key];
                updateProgress();
                renderQuiz(currentTabIndex); 
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }
        }

        // 안전하게 초기화 실행
        window.addEventListener('DOMContentLoaded', () => {
            parseRawData();
            renderTabs();
            renderQuiz(currentTabIndex);
            updateTabsStyle();
        });

    </script>
</body>
</html>
